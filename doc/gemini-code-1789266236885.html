Virtqueue 実装・学習ハンドブック

virtio のコア技術である Virtqueue（共有メモリ上のリングバッファ） の構造を、C言語の実装を通して段階的に学ぶための資料です。QNXの vdev や Linuxカーネルの virtio ドライバの動作原理を、Mac上の共有メモリ（POSIX shm）を用いて理解します。

Step 1: Virtqueue（Split Virtqueue）の3大構造

virtio 1.0の基本仕様（Split Queue）では、ホストとゲストが共有するメモリ領域は以下の3つの配列・構造体で成り立っています。

[ 共有メモリ領域 (Shared Memory) ] 1. Descriptor Table : バッファの実体（アドレス、サイズ、属性）を管理する表 2. Available Ring : ゲストが「これ処理して！」とホストに渡すキュー（IDの配列） 3. Used Ring : ホストが「処理終わったよ！」とゲストに返すキュー（IDの配列）

1. Descriptor Table（ディスクリプタ配列）

1つのデータ領域（パケットやブロックバッファ）の位置と長さを表します。複数つなげる（チェイン）ことも可能です。

struct vring_desc {
    uint64_t addr;   // バッファのアドレス（ゲスト物理アドレス、今回は共有メモリ内オフセット）
    uint32_t len;    // バッファのバイト長
    uint16_t flags;  // 次のディスクリプタへ続くか(VRING_DESC_F_NEXT)や読み書き権限
    uint16_t next;   // flagsにNEXTが立っている場合、次のDescriptorのインデックス
};

2. Available Ring（ゲスト → ホスト）

ゲストOSがホストへ「処理待ちリクエスト」を登録するリングバッファです。

struct vring_avail {
    uint16_t flags;            // 通知を抑制するかどうかのフラグ
    uint16_t idx;              // 次に書き込むリングの位置（単調増加カウンタ）
    uint16_t ring[QUEUE_SIZE]; // 処理してほしい Descriptor のインデックス一覧
};

3. Used Ring（ホスト → ゲスト）

ホスト（QEMUやQNXのvdev）が「処理完了」をゲストOSに報告するリングバッファです。

struct vring_used_elem {
    uint32_t id;  // 完了した Descriptor の先頭インデックス
    uint32_t len; // 実際に書き込んだ/読み取ったバイト数
};

struct vring_used {
    uint16_t flags;
    uint16_t idx; // 次に書き込むリングの位置（単調増加カウンタ）
    struct vring_used_elem ring[QUEUE_SIZE];
};

Step 2: C言語による最小Virtqueueシミュレータ

Macのターミナル上で、POSIX共有メモリ（shm） を使って「Guest役（送信）」と「Host役（受信）」を2つのプロセスで動かします。

virtio_mini.h 
#ifndef VIRTIO_MINI_H
#define VIRTIO_MINI_H

#include <stdint.h>

#define QUEUE_SIZE 8
#define QUEUE_MASK (QUEUE_SIZE - 1) /* Virtio仕様: キューサイズは必ず2のべき乗 */
#define SHM_NAME "/virtio_shm_demo"
#define DATA_BUF_SIZE 256

/* ディスクリプタ */
struct vring_desc {
    uint64_t offset; // 共有メモリ先頭からのデータバッファオフセット (実機ではゲスト物理アドレスGPA)
    uint32_t len;
    uint16_t flags;
    uint16_t next;
};

/* Available Ring (ゲスト -> ホスト) */
struct vring_avail {
    uint16_t flags;
    volatile uint16_t idx; // コンパイラのレジスタ最適化を抑止
    uint16_t ring[QUEUE_SIZE];
};

/* Used Ring (ホスト -> ゲスト) */
struct vring_used_elem {
    uint32_t id;
    uint32_t len;
};

struct vring_used {
    uint16_t flags;
    volatile uint16_t idx; // コンパイラのレジスタ最適化を抑止
    struct vring_used_elem ring[QUEUE_SIZE];
};

/* 共有メモリ全体マップ */
struct shared_virtqueue_layout {
    struct vring_desc  desc[QUEUE_SIZE];
    struct vring_avail avail;
    struct vring_used  used;
    uint8_t data_payloads[QUEUE_SIZE][DATA_BUF_SIZE]; // データ本体領域
};

#endif
host.c 
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <sys/mman.h>
#include <unistd.h>
#include "virtio_mini.h"

int main() {
    int fd = shm_open(SHM_NAME, O_CREAT | O_RDWR, 0666);
    ftruncate(fd, sizeof(struct shared_virtqueue_layout));
    
    struct shared_virtqueue_layout *vq = mmap(
        NULL, sizeof(struct shared_virtqueue_layout),
        PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0
    );

    printf("[Host Backend] Virtqueue Ready. Waiting for Guest...\n");

    uint16_t last_avail_idx = 0;

    while (1) {
        // Available Ringのidxが進んだかを監視 (実機ではDoorbellトラップ等で起床)
        if (vq->avail.idx != last_avail_idx) {
            __sync_synchronize(); // Read Barrier: idx読込後にリングやディスクリプタを読み込む (ARM弱順序対策)
            uint16_t desc_idx = vq->avail.ring[last_avail_idx & QUEUE_MASK];
            
            // ディスクリプタを参照してデータを取り出す
            char *msg = (char *)((uintptr_t)vq + vq->desc[desc_idx].offset);
            printf("[Host] Received via Desc[%u]: %s\n", desc_idx, msg);

            // 処理完了を Used Ring に記録 (ビットマスクで計算)
            uint16_t used_slot = vq->used.idx & QUEUE_MASK;
            vq->used.ring[used_slot].id = desc_idx;
            vq->used.ring[used_slot].len = 0; // 送信バッファ返却のため書込長は0 (Virtio仕様準拠)
            
            __sync_synchronize(); // Write Barrier: used.ring書込を反映した後にused.idxを進める
            vq->used.idx++;       // ホスト側進行カウンタをインクリメント

            last_avail_idx++;
        }
        usleep(10000); // 10msスリープ（擬似ポーリング）
    }

    munmap(vq, sizeof(struct shared_virtqueue_layout));
    close(fd);
    return 0;
}
guest.c 
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <sys/mman.h>
#include <unistd.h>
#include "virtio_mini.h"

int main() {
    int fd = shm_open(SHM_NAME, O_RDWR, 0666);
    if (fd < 0) {
        perror("shm_open failed (Run host first)");
        exit(1);
    }

    struct shared_virtqueue_layout *vq = mmap(
        NULL, sizeof(struct shared_virtqueue_layout),
        PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0
    );

    uint16_t last_used_idx = vq->used.idx;
    uint32_t slot = 0; // 今回はスロット0を使用

    // 1. データペイロードに書き込み
    char *target = (char *)vq->data_payloads[slot];
    snprintf(target, DATA_BUF_SIZE, "Hello from Guest! (Packet #1)");

    // 2. ディスクリプタ設定
    vq->desc[slot].offset = (uintptr_t)vq->data_payloads[slot] - (uintptr_t)vq;
    vq->desc[slot].len = strlen(target) + 1;
    vq->desc[slot].flags = 0; // 単一バッファ（チェインなし）

    // 3. Available Ring に追加 (ビットマスクでインデックス計算)
    uint16_t avail_slot = vq->avail.idx & QUEUE_MASK;
    vq->avail.ring[avail_slot] = slot;
    
    __sync_synchronize(); // Write Barrier: バッファ/Desc/Ring書込を反映した後にidxを進める
    vq->avail.idx++;      // ゲスト側進行カウンタをインクリメント
    printf("[Guest] Submitted buffer (slot %u). Waiting for Host to process...\n", slot);

    // 4. Used Ring が返ってくるのを待機 (実機では割り込み待機)
    while (vq->used.idx == last_used_idx) {
        usleep(1000);
    }
    __sync_synchronize(); // Read Barrier: used.idx読込後にused.ringを読み出す (ARM弱順序対策)

    printf("[Guest] Host marked Desc[%u] as USED! I/O Complete.\n",
           vq->used.ring[last_used_idx & QUEUE_MASK].id);

    munmap(vq, sizeof(struct shared_virtqueue_layout));
    close(fd);
    return 0;
}

コンパイルと動作確認

ターミナルを2つ開いて実行します。

# ターミナル 1 (Host側)
clang host.c -o host
./host

# ターミナル 2 (Guest側)
clang guest.c -o guest
./guest
ここがポイント（Virtio仕様と実機設計の要点）
  • 単調増加カウンタとビットマスク（& (QUEUE_SIZE - 1)）: idx は uint16_t（0〜65535）として単調増加し、オーバーフロー時に 65535 → 0 へ自然に周回（ラップアラウンド）します。 キューサイズが 2のべき乗（例: 8）であるため、idx & (QUEUE_SIZE - 1) を用いることで、ラップアラウンド境界を跨いでもスロット番号が不連続にならず安全に巡回できます。また、CPUの除算命令（div）を回避し、1サイクルのビットマスクで高速処理されます。
  • メモリバリアの対称性（Read / Write Barrier）: ARM64（Apple Siliconや車載SoC）はアウトオブオーダー実行を行う メモリ弱順序（Weak Ordering） です。 書き込み側（WMB）だけでなく、読み込み側（RMB）でも「idx の更新を確認してからリングやバッファを参照する」ための Load-Load バリア（Read Barrier）が不可欠です。現代のC11/C++11では std::atomic_thread_fence(memory_order_acquire / release) や stlr/ldar 命令がこれに対応します。
  • キャッシュライン偽共有（False Sharing）の防止: 実機Virtio仕様では、ゲストが書く avail とホストが書く used は物理的に別々のページやキャッシュライン（ARM64: 64〜128バイト）に分離して配置され、両コア間のキャッシュ無効化競合を防ぎます。

Step 3: Linuxカーネルコードとの対応関係

自作シミュレータのコードは、Linuxカーネル（drivers/virtio/virtio_ring.c）の標準実装と以下のように対応しています。

自作コードの操作 役割 実際のLinuxカーネル関数 / マクロ
vq->desc[slot] = ... バッファ情報の登録 virtqueue_add_outbuf() / virtqueue_add_sgs()
vq->avail.ring[...] = ... 処理キューへの投入 virtqueue_add() 内部の Avail Ring 更新処理
vq->avail.idx++ 送信コミット 同上（メモリバリア virtio_wmb() 直後に実行）
Doorbell（今回は省略） ホストへの通知 virtqueue_kick()（PCIレジスタ書込によるVM-Exit）

初学者が陥りやすい7大誤解と注意点（Pitfalls & Misconceptions）

virtio の学習や実装において、抽象化レイヤの境界や仮想化特有の挙動を見誤ると、重大なバグや設計ミスにつながります。特に初学者が勘違いしやすい重要ポイントを整理しました。

⚠️ 誤解 1: 「virtio はソケットやネットワーク通信の一種である」

✕ よくある勘違い: virtio を TCP や Unix ドメインソケットのような IPC プロトコルだと思っている。
◯ 正しい理解: virtio の本質は 「ゲストとホストが共有メモリ（DMAバッファ＋リングバッファ）を介してやり取りするデバイスI/O」 です。ネットワークスタックは一切通りません。本サンプルのホスト側（src/host.cpp）が Unix ドメインソケットに接続しているのは、QEMU がキャラクタデバイスのバックエンド（chardev socket）としてソケットを公開しているためであり、virtio 仕様自体の要件ではありません。

⚠️ 誤解 2: 「アプリケーションが直接 virtqueue（リング）を操作する」

✕ よくある勘違い: C++アプリの中でディスクリプタを登録したり virtqueue_kick を呼ぶコードを書く必要がある。
◯ 正しい理解: 通常、Virtqueue やリングバッファ、PCI割り込みを操作するのは Linuxカーネル内のドライバー（Frontend） の役目です。アプリケーションは通常のファイルと同じように /dev/virtio-ports/... に対して標準の open(), read(), write() を呼ぶだけで通信できます（DPDK などのユーザー空間ドライバーを使う特殊なケースを除く）。

⚠️ 誤解 3: 「write() が成功したらホストの処理や返信が完了している」

✕ よくある勘違い: write() が正常終了したのだから、ホスト側もデータを受け取って処理を終えているはずだ。
◯ 正しい理解: ゲストの write() 成功は「カーネルの送信バッファ／送信virtqueueにデータが登録された」ことしか意味しません。ホスト側の処理完了や返信の受領は、ホストが Used Ring にバッファを返却し、ゲストOSへの完了通知（仮想割り込み）を経て read() で取得する 非同期パイプライン です。

⚠️ 誤解 4: 「ホストとゲスト間でポインタ（メモリアドレス）を直接渡せる」

✕ よくある勘違い: ゲスト側で void *buf = malloc(100); して、そのポインタ値をホストに渡せばホストが読める。
◯ 正しい理解: ゲスト仮想アドレス（GVA）とホスト仮想アドレス（HVA）は完全に独立したアドレス空間です。ディスクリプタに書き込むのは ゲスト物理アドレス（GPA）（または IOMMU/Stage-2 変換下の IOVA）です。ホスト側（QEMUやハイパーバイザー）は、Stage-2 ページテーブル変換を通じて GPA を自身の HVA にマップして初めてバッファへアクセスできます。ポインタ値をそのまま渡してもホスト側では不正メモリアクセスになります。

⚠️ 誤解 5: 「キック（Doorbell / 通知）はタダ（軽量）である」

✕ よくある勘違い: 1パケット送るたびに毎回 kick を呼んでもパフォーマンスに影響はない。
◯ 正しい理解: ゲストからホストへのキックは、PCI/MMIO レジスタ書き込みに伴う VM-Exit（CPU仮想化トラップ） を引き起こします。VM-Exit は通常のメモリアクセスに比べて数千倍重く、CPU パイプラインをストールさせます。そのため Virtio 仕様には VIRTIO_F_EVENT_IDX や VRING_AVAIL_F_NO_INTERRUPT などの通知抑制フラグがあり、複数バッファをまとめて1回だけキックするバッチ処理が実機性能の鍵となります。

⚠️ 誤解 6: 「シリアルやキャラクタデバイスでもパケット（メッセージ）境界が保たれる」

✕ よくある勘違い: 10バイトの write() を呼んだら、相手側も必ず 1回の read() でちょうど10バイト受け取れる。
◯ 正しい理解: virtio-console やキャラクターデバイスは バイトストリーム であり、メッセージ境界（パケットの区切り）を保存しません。ネットワークの TCP と同様に、途中で分割されたり複数メッセージが結合して届く可能性があります。そのため src/transport.hpp のように、アプリケーション層で「改行区切り」や「ヘッダ（長さ）＋本文」などのフレーミングプロトコルを実装する必要があります。

⚠️ 誤解 7: 「仮想マシンなのでマルチコアのメモリバリアは不要である」

✕ よくある勘違い: 仮想化環境のメモリなので、単一スレッドのように順番通りに読み書きされるはずだ。
◯ 正しい理解: ゲスト vCPU とホスト CPU（QEMUスレッドなど）は、物理的に異なる CPU コア上で並行動作しています。特に ARM64（Apple Silicon や 車載SoC）はアウトオブオーダー実行を行う 弱順序（Weak Ordering） CPU です。バッファを書いてから idx を進める Store-Store バリア（WMB）と、idx を検知してからバッファを読む Load-Load バリア（RMB）を正しく配置しないと、相手側で古いゴミデータを参照してしまうデータ競合が発生します。
